    Create `src/shortfire/observability/metrics.py` — adapted from RESEARCH.md §Pattern 5 but extended for the 4 base metrics per UI-SPEC §Metric registry:
    ```python
    from typing import NamedTuple
    from prometheus_client import Counter, Histogram, Gauge, CollectorRegistry, generate_latest, CONTENT_TYPE_LATEST
    from fastapi import FastAPI, Response

    REGISTRY = CollectorRegistry()

    LATENCY_BUCKETS = (0.005, 0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1.0, 2.5, 5.0, 10.0)  # UI-SPEC §Metric registry

    class ServiceMetrics(NamedTuple):
        http_requests_total: Counter
        http_request_duration_seconds: Histogram
        service_event_emitted_total: Counter
        build_info: Gauge

    def build_metrics_for_service(service_name: str) -> ServiceMetrics:
        """Per-service metric family factory. service_name must be snake_case (UI-SPEC §Metric naming convention)."""
        svc = service_name.replace("-", "_")  # kebab -> snake (UI-SPEC: Prometheus forbids hyphens in metric names)
        return ServiceMetrics(
            http_requests_total=Counter(
                f"shortfire_{svc}_http_requests_total",
                "HTTP request count",
                ["method", "path", "status"],
                registry=REGISTRY,
            ),
            http_request_duration_seconds=Histogram(
                f"shortfire_{svc}_http_request_duration_seconds",
                "HTTP request latency in seconds",
                ["method", "path"],
                buckets=LATENCY_BUCKETS,
                registry=REGISTRY,
            ),
            service_event_emitted_total=Counter(
                f"shortfire_{svc}_service_event_emitted_total",
                "service_event rows emitted",
                ["event_type"],
                registry=REGISTRY,
            ),
            build_info=Gauge(
                f"shortfire_{svc}_build_info",
                "Build info (always 1)",
                ["version", "commit_sha", "env"],
                registry=REGISTRY,
            ),
        )

    def install_metrics_endpoint(app: FastAPI) -> None:
        @app.get("/metrics")
        def metrics() -> Response:
            return Response(content=generate_latest(REGISTRY), media_type=CONTENT_TYPE_LATEST)
    ```
