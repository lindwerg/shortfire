    ```
    import asyncio
    from datetime import datetime, timezone, timedelta
    from decimal import Decimal
    import pytest
    from sqlalchemy import text
    from sqlalchemy.ext.asyncio import create_async_engine

    from shortfire.ingest.mexc.backfill import backfill_universe
    from tests.fakes.mexc import FakeMexcClient


    @pytest.mark.integration
    @pytest.mark.asyncio
    @pytest.mark.timeout(600)  # 10 min hard cap — D-94 target is 5 min
    async def test_six_day_universe_backfill_completes_under_ci_budget(migrated_db: str):
        """STOR-08 sanity slice (D-94): 50 symbols × 1m × 6 days through the full backfill pipeline.

        Asserts:
          - backfill_universe completes within the budget
          - Final row count in raw_mexc_candles_1m matches expectation (within 1%)
          - source='mexc_native' on every row
          - Re-running produces zero new rows (idempotency, DATA-09)
        """
        engine = create_async_engine(migrated_db.replace("postgresql://", "postgresql+asyncpg://"), pool_size=10, pool_pre_ping=True)
        # TRUNCATE for test isolation
        async with engine.begin() as conn:
            await conn.execute(text("TRUNCATE TABLE raw_mexc_candles_1m"))

        # Build a FakeMexcClient with 50 synthetic symbols, 6-day 1m coverage
        symbols = [f"SYM{i:02d}/USDT:USDT" for i in range(50)]
        since = datetime.now(timezone.utc) - timedelta(days=6)
        until = datetime.now(timezone.utc)
        fake = FakeMexcClient.with_synthetic_candles(symbols=symbols, timeframe="1m", since=since, until=until)

        # Run backfill_universe
        results = await backfill_universe(engine, fake, symbols, timeframes=["1m"], since=since, until=until, max_concurrency=8)
        total_inserted = sum(results.values())
        assert total_inserted > 0

        # Verify row count
        async with engine.connect() as conn:
            count_row = await conn.execute(text("SELECT count(*) FROM raw_mexc_candles_1m"))
            count = count_row.scalar()
        expected_minimum = 50 * 8640 * 0.99  # within 1% of perfect coverage
        assert count >= expected_minimum, f"Expected ~432K rows, got {count}"

        # Verify source attribution
        async with engine.connect() as conn:
            bad_sources = (await conn.execute(text(
                "SELECT count(*) FROM raw_mexc_candles_1m WHERE source != 'mexc_native'"
            ))).scalar()
        assert bad_sources == 0

        # Verify idempotency (DATA-09): re-run, assert count unchanged
        await backfill_universe(engine, fake, symbols, timeframes=["1m"], since=since, until=until, max_concurrency=8)
        async with engine.connect() as conn:
            count2 = (await conn.execute(text("SELECT count(*) FROM raw_mexc_candles_1m"))).scalar()
        assert count2 == count, f"Re-ingest produced duplicates: was {count}, now {count2}"

        await engine.dispose()
    ```
