    ```
    from datetime import date, datetime, timezone
    from decimal import Decimal
    import structlog
    from sqlalchemy import select, update, text
    from sqlalchemy.ext.asyncio import AsyncEngine, AsyncSession

    from shortfire.ingest.storage.copy import copy_into_hypertable
    from shortfire.ingest.dead_letter.writer import write_to_dead_letter
    from shortfire.observability.events import assert_event_registered
    from shortfire.observability.metrics_data_platform import build_data_platform_metrics
    from shortfire.db.models.symbols import Symbol
    from shortfire.ingest.mexc.client import MexcClient
    from shortfire.ingest.coingecko.client import CoinGeckoClient

    log = structlog.get_logger("ingest.universe.snapshot")

    QUALIFYING_VOLUME_USD = Decimal("500000")  # UNIV-01 threshold


    async def universe_snapshot_job(
        engine: AsyncEngine,
        mexc_client: MexcClient,
        coingecko_client: CoinGeckoClient | None = None,
    ) -> int:
        """Daily universe snapshot at 00:05 UTC (D-77).

        - Read MEXC ticker universe, filter quote_volume_usd > $500K
        - Write one row per symbol into universe_snapshots (append-only)
        - Diff against yesterday for new-listing detection (UNIV-04)
        - Soft-delete symbols absent today via symbols.delisted_at = now()
        - Optionally enrich new symbols via CoinGecko /coins/{id}

        Returns rows written into universe_snapshots.
        """
        snapshot_date = date.today()
        # 1. Fetch live tickers
        try:
            await mexc_client._client.load_markets()
            tickers = await mexc_client._client.fetch_tickers()
        except Exception as exc:
            await write_to_dead_letter("mexc_native", "fetch_tickers", None, "", type(exc).__name__, str(exc)[:2000])
            log.error("universe.snapshot.fetch_tickers_failed", exc_info=exc)
            return 0

        # 2. Filter qualifying universe (UNIV-01)
        qualifying: dict[str, dict] = {}
        for sym, ticker in tickers.items():
            if not sym.endswith(":USDT"):  # USDT-perp only
                continue
            qv = ticker.get("quoteVolume", 0)
            try:
                qv_usd = Decimal(str(qv))
            except Exception:
                continue
            is_qualifying = qv_usd > QUALIFYING_VOLUME_USD
            qualifying[sym] = {
                "volume_24h_usd": qv_usd,
                "price_usd": Decimal(str(ticker.get("last", 0))) if ticker.get("last") else None,
                "is_qualifying": is_qualifying,
            }

        # 3. Write snapshot rows (UNIV-02, STOR-06)
        records = [
            (snapshot_date, sym, info["volume_24h_usd"], info["price_usd"], info["is_qualifying"], "mexc_native", "ok")
            for sym, info in qualifying.items()
        ]
        n = await copy_into_hypertable(
            engine, "universe_snapshots", records,
            columns=("snapshot_date", "symbol", "volume_24h_usd", "price_usd", "is_qualifying", "source", "quality_flag"),
            conflict_columns=("snapshot_date", "symbol"),
        )

        # 4. Diff against yesterday for new-listing + delisting (UNIV-04, STOR-07)
        async with engine.connect() as conn:
            yesterday_rows = await conn.execute(text(
                "SELECT symbol FROM universe_snapshots WHERE snapshot_date = :y"
            ), {"y": snapshot_date - __import__("datetime").timedelta(days=1)})
            yesterday_set = {r[0] for r in yesterday_rows}
        today_set = set(qualifying.keys())
        new_symbols = today_set - yesterday_set
        delisted_symbols = yesterday_set - today_set

        async with AsyncSession(engine) as session:
            for sym in new_symbols:
                assert_event_registered("universe.symbol.new")
                log.info("universe.symbol.new", symbol=sym, snapshot_date=str(snapshot_date))
                native = sym.replace("/USDT:USDT", "_USDT")  # ccxt → MEXC native
                stmt = text("""
                    INSERT INTO symbols (symbol, exchange, market_type, mexc_native_symbol, first_seen_at, tier)
                    VALUES (:symbol, 'mexc', 'swap', :native, now(), 2)
                    ON CONFLICT (symbol) DO UPDATE SET delisted_at = NULL
                """)
                await session.execute(stmt, {"symbol": sym, "native": native})
            for sym in delisted_symbols:
                assert_event_registered("universe.symbol.delisted")
                log.info("universe.symbol.delisted", symbol=sym, snapshot_date=str(snapshot_date))
                await session.execute(text("UPDATE symbols SET delisted_at = now() WHERE symbol = :s AND delisted_at IS NULL"), {"s": sym})
            await session.commit()

        # 5. Update universe_symbols_count gauge
        metrics = build_data_platform_metrics()
        metrics.universe_symbols_count.labels(status="qualifying").set(sum(1 for v in qualifying.values() if v["is_qualifying"]))
        metrics.universe_symbols_count.labels(status="non_qualifying").set(sum(1 for v in qualifying.values() if not v["is_qualifying"]))

        assert_event_registered("universe.snapshot.created")
        log.info("universe.snapshot.created", snapshot_date=str(snapshot_date), n_qualifying=sum(1 for v in qualifying.values() if v["is_qualifying"]), n_new=len(new_symbols), n_delisted=len(delisted_symbols))
        return n


    async def universe_at(engine: AsyncEngine, snapshot_date: date) -> set[str]:
        """Point-in-time query — UNIV-03 invariant. Returns symbols qualifying AT snapshot_date."""
        async with engine.connect() as conn:
            rows = await conn.execute(text(
                "SELECT symbol FROM universe_snapshots WHERE snapshot_date = :d AND is_qualifying = TRUE"
            ), {"d": snapshot_date})
            return {r[0] for r in rows}
    ```

    **Step 4: Create `src/shortfire/ingest/universe/tier1.py`** per D-46 + `<specifics>` §1:

    ```
    async def recompute_tier1(engine: AsyncEngine, top_n: int = 50) -> None:
        """Rerank symbols by last-7d volume; top-N → tier=1, rest → tier=2 (D-46)."""
        async with engine.connect() as conn:
            rows = await conn.execute(text("""
                WITH vol AS (
                    SELECT symbol, SUM(volume) AS v7d
                    FROM raw_mexc_candles_1d
                    WHERE ts >= now() - INTERVAL '7 days'
                    GROUP BY symbol
                )
                SELECT symbol FROM vol ORDER BY v7d DESC LIMIT :n
            """), {"n": top_n})
            top_set = {r[0] for r in rows}
        async with engine.begin() as conn:
            await conn.execute(text("UPDATE symbols SET tier = 1 WHERE symbol = ANY(:syms)"), {"syms": list(top_set)})
            await conn.execute(text("UPDATE symbols SET tier = 2 WHERE symbol NOT IN (SELECT unnest(:syms)) AND delisted_at IS NULL"), {"syms": list(top_set)})
        log.info("universe.tier1.recomputed", n_tier1=len(top_set))
    ```

    ```python
    # src/shortfire/ingest/state/kv_state.py
    """Round-robin cursor persistence via ingest_runs.kv_state JSONB column (RESEARCH.md §4.6).

    Every Coinglass and MEXC round-robin scheduler job uses this helper to read/write its
    `last_symbol_index` cursor across scheduler restarts. Storing in ingest_runs (a hypertable)
    instead of a separate `kv_store` table keeps the schema surface minimal — D-77 cron jobs
    already write into ingest_runs on each completion, so this just adds a read path.

    Schema reminder (from migration 0013): ingest_runs has columns
        (id, ts, source, dataset, job_id, started_at, finished_at, status, rows_written, error_msg, kv_state JSONB, quality_flag, ingested_at)
    Each job_id may have many rows; `load_kv_state` returns the latest `kv_state` value for that job_id
    (ordered by ts DESC LIMIT 1). `save_kv_state` INSERTs a new row with the updated state.
    """
    from datetime import datetime, timezone
    from typing import Any
    import json
    import uuid
    import structlog
    from sqlalchemy import text
    from sqlalchemy.ext.asyncio import AsyncEngine

    log = structlog.get_logger("ingest.state.kv_state")


    async def load_kv_state(engine: AsyncEngine, job_id: str) -> dict[str, Any]:
        """Return the latest persisted kv_state for `job_id`, or {} if none recorded."""
        async with engine.connect() as conn:
            row = (await conn.execute(text(
                "SELECT kv_state FROM ingest_runs WHERE job_id = :j AND kv_state IS NOT NULL ORDER BY ts DESC LIMIT 1"
            ), {"j": job_id})).first()
        if row is None or row[0] is None:
            return {}
        # asyncpg returns JSONB as Python dict directly; SQLAlchemy passes it through
        return row[0] if isinstance(row[0], dict) else json.loads(row[0])


    async def save_kv_state(engine: AsyncEngine, job_id: str, state: dict[str, Any]) -> None:
        """Insert a new ingest_runs row recording this job's updated kv_state cursor."""
        now = datetime.now(timezone.utc)
        # Derive source from job_id prefix (e.g. "coinglass.oi" → source="coinglass_aggregate")
        if job_id.startswith("coinglass"):
            source = "coinglass_aggregate"
        elif job_id.startswith("coingecko"):
            source = "coingecko"
        elif job_id.startswith("mexc"):
            source = "mexc_native"
        else:
            source = "mexc_native"  # default
        # Derive dataset from job_id suffix (e.g. "coinglass.oi" → dataset="oi")
        dataset = job_id.split(".", 1)[1] if "." in job_id else job_id
        async with engine.begin() as conn:
            await conn.execute(text("""
                INSERT INTO ingest_runs (id, ts, source, dataset, job_id, started_at, finished_at, status, rows_written, kv_state, quality_flag)
                VALUES (:id, :ts, :src, :ds, :jid, :ts, :ts, 'succeeded', 0, :st, 'ok')
            """), {
                "id": uuid.uuid4(),
                "ts": now,
                "src": source,
                "ds": dataset,
                "jid": job_id,
                "st": json.dumps(state),
            })
        log.debug("kv_state.saved", job_id=job_id, state_keys=list(state.keys()))
    ```
