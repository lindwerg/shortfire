    from datetime import date, datetime
    from decimal import Decimal
    from typing import Any, Literal
    from pydantic import BaseModel, ConfigDict, Field


    class CoinsMarketsRow(BaseModel):
        """One row from /coins/markets endpoint. Pitfall G `extra='allow'` for forward-compat."""
        model_config = ConfigDict(frozen=True, strict=True, extra="allow")

        id: str
        symbol: str  # bare-coin form e.g. "btc"
        name: str
        current_price: Decimal | None = None
        market_cap: Decimal | None = None
        total_volume: Decimal | None = None
        last_updated: datetime | None = None
        raw_payload: dict[str, Any] | None = None  # retained for forensics; populated by from_raw classmethod

        @classmethod
        def from_raw(cls, raw: dict[str, Any]) -> "CoinsMarketsRow":
            """Construct from /coins/markets row, retaining the raw dict in raw_payload."""
            # Coerce numeric-or-null fields explicitly to Decimal
            data = {k: raw[k] for k in ("id","symbol","name","current_price","market_cap","total_volume","last_updated") if k in raw}
            return cls.model_validate({**data, "raw_payload": raw})

        def to_record(self, ts: datetime, symbol_unified: str | None = None) -> tuple:
            """Build the 9-tuple for `raw_coingecko_market` INSERT.

            Args:
                ts: snapshot timestamp (caller passes datetime.now(timezone.utc) for the daily run)
                symbol_unified: optional override — if None, falls back to the bare-coin symbol uppercased.
            """
            sym = symbol_unified or self.symbol.upper()
            return (
                sym,
                ts,
                self.current_price,
                self.total_volume,
                self.market_cap,
                None,  # category — populated by CoinDetailResponse if available
                None,  # listing_date — populated by CoinDetailResponse if available
                self.raw_payload,
                "coingecko",
                "ok",
            )


    class CoinDetailResponse(BaseModel):
        """/coins/{id} response — used for new-listing category + listing_date enrichment (D-55)."""
        model_config = ConfigDict(frozen=True, strict=True, extra="allow")

        id: str
        symbol: str
        name: str
        categories: tuple[str, ...] = ()
        genesis_date: date | None = None  # CoinGecko uses this for "listing_date"
        market_data: dict[str, Any] | None = None
        raw_payload: dict[str, Any] | None = None

        @classmethod
        def from_raw(cls, raw: dict[str, Any]) -> "CoinDetailResponse":
            data = {k: raw[k] for k in ("id","symbol","name","categories","genesis_date","market_data") if k in raw}
            # categories may arrive with None inside; filter
            if "categories" in data and isinstance(data["categories"], list):
                data["categories"] = tuple(c for c in data["categories"] if c is not None)
            return cls.model_validate({**data, "raw_payload": raw})
    ```

    import httpx
    import structlog
    from pydantic import ValidationError

    from shortfire.ingest.retry import coingecko_retry
    from shortfire.ingest.rate_limit import COINGECKO_LIMITER
    from shortfire.ingest.dead_letter.writer import write_to_dead_letter
    from shortfire.ingest.coingecko.schemas import CoinsMarketsRow, CoinDetailResponse
    from shortfire.settings.data_platform import DataPlatformSettings

    log = structlog.get_logger("ingest.coingecko.client")

    _COINGECKO_BASE = "https://api.coingecko.com/api/v3"


    class CoinGeckoClient:
        """Concrete CoinGecko Demo-tier client. D-36 ~$35/mo Demo; D-56 28/60 limiter; D-55 universe-only."""

        def __init__(self, settings: DataPlatformSettings) -> None:
            if settings.coingecko is None:
                raise RuntimeError("DataPlatformSettings.coingecko must be configured for ingest")
            self._api_key = settings.coingecko.api_key.get_secret_value()
            self._client = httpx.AsyncClient(
                http2=True,
                timeout=10.0,
                limits=httpx.Limits(max_connections=20, max_keepalive_connections=10),
                headers={"x-cg-demo-api-key": self._api_key, "accept": "application/json"},
            )

        async def _call(self, endpoint: str, params: dict | None = None) -> Any | None:
            url = f"{_COINGECKO_BASE}{endpoint}"
            async with COINGECKO_LIMITER:
                r = await self._client.get(url, params=params)
            if r.status_code in (429, 500, 502, 503, 504):
                r.raise_for_status()
            if r.status_code >= 400:
                await write_to_dead_letter("coingecko", endpoint, params.get("id") if params else None,
                                           r.text[:8000], "HTTPStatusError", f"HTTP {r.status_code}")
                return None
            return r.json()

        @coingecko_retry
        async def fetch_coins_markets(self, vs_currency: str = "usd", per_page: int = 250, page: int = 1) -> tuple[CoinsMarketsRow, ...] | None:
            payload = await self._call("/coins/markets", params={"vs_currency": vs_currency, "per_page": per_page, "page": page, "order": "market_cap_desc"})
            if payload is None:
                return None
            try:
                # /coins/markets returns a JSON array directly
                if not isinstance(payload, list):
                    raise ValidationError.from_exception_data("CoinsMarkets", [{"type":"list_type"}])
                return tuple(CoinsMarketsRow.from_raw(raw) for raw in payload)
            except (ValidationError, ValueError, TypeError) as exc:
                await write_to_dead_letter("coingecko", "/coins/markets", None, str(payload)[:8000], "ValidationError", str(exc)[:2000])
                return None

        @coingecko_retry
        async def fetch_coin_detail(self, coin_id: str) -> CoinDetailResponse | None:
            payload = await self._call(f"/coins/{coin_id}", params={"id": coin_id, "localization": "false", "tickers": "false", "community_data": "false", "developer_data": "false"})
            if payload is None:
                return None
            try:
                return CoinDetailResponse.from_raw(payload)
            except (ValidationError, ValueError, TypeError) as exc:
                await write_to_dead_letter("coingecko", f"/coins/{coin_id}", coin_id, str(payload)[:8000], "ValidationError", str(exc)[:2000])
                return None

        async def close(self) -> None:
            await self._client.aclose()
    ```

    **Step 1: Create `src/shortfire/ingest/coingecko/universe.py`**:

    ```
    from datetime import datetime, timezone
    import structlog
    from sqlalchemy.ext.asyncio import AsyncEngine

    from shortfire.ingest.storage.copy import copy_into_hypertable
    from shortfire.ingest.coingecko.client import CoinGeckoClient

    log = structlog.get_logger("ingest.coingecko.universe")


    async def fetch_and_write_daily(
        engine: AsyncEngine,
        client: CoinGeckoClient,
        ts: datetime | None = None,
    ) -> int:
        """Daily CoinGecko universe-metadata refresh — D-55, D-77 (00:30 UTC cron).

        Fetches the top-250 USD-volume coins from /coins/markets and writes one row per coin
        into raw_coingecko_market. Idempotent via PK (symbol, ts, source).

        Args:
            engine: AsyncEngine.
            client: CoinGeckoClient.
            ts: Snapshot timestamp (default: now(UTC)). Use the same `ts` across the entire
                daily batch so all rows share a single snapshot moment.

        Returns:
            Rows written (after ON CONFLICT DO NOTHING).
        """
        if ts is None:
            ts = datetime.now(timezone.utc)
        rows = await client.fetch_coins_markets(vs_currency="usd", per_page=250, page=1)
        if rows is None:
            log.warning("coingecko.universe.skipped", reason="fetch returned None — dead_letter already logged")
            return 0
        records = [r.to_record(ts) for r in rows]
        n = await copy_into_hypertable(
            engine,
            "raw_coingecko_market",
            records,
            columns=("symbol", "ts", "price_usd", "volume_24h_usd", "market_cap_usd", "category", "listing_date", "raw_payload", "source", "quality_flag"),
            conflict_columns=("symbol", "ts", "source"),
        )
        log.info("coingecko.universe.completed", rows_written=n)
        return n
    ```
