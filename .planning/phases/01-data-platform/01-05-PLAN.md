    Module docstring cites D-40 (defaultType=swap; symbol convention BTC/USDT:USDT), D-41 (ccxt 4.5 pin), D-49 (freshness contract), Pitfall 8 (never log signing material → verbose=False).

    ```
    import ccxt.pro as ccxtpro
    import structlog
    from datetime import datetime, timezone
    from typing import Self
    import asyncio
    from shortfire.ingest.retry import mexc_retry
    from shortfire.ingest.rate_limit import MEXC_LIMITER
    from shortfire.ingest.mexc.schemas import MexcOhlcvRow, MexcFundingRow, MexcOpenInterestRow, MexcTradeRow
    from shortfire.ingest.dead_letter.writer import write_to_dead_letter
    from shortfire.settings.data_platform import DataPlatformSettings
    from shortfire.domain.market import Candle, Funding, OrderBook, OrderBookLevel
    from shortfire.domain.trading import Order
    from decimal import Decimal

    log = structlog.get_logger("ingest.mexc.client")


    def build_mexc_swap_client(settings: DataPlatformSettings) -> "MexcClient":
        if settings.mexc is None:
            raise RuntimeError("DataPlatformSettings.mexc must be configured for live ingest (set MEXC__READ_KEY + MEXC__READ_SECRET)")
        raw = ccxtpro.mexc({
            "apiKey": settings.mexc.read_key.get_secret_value(),
            "secret": settings.mexc.read_secret.get_secret_value(),
            "enableRateLimit": True,
            "options": {"defaultType": "swap", "recvWindow": 10000},
        })
        raw.verbose = False  # Pitfall 8 — never log signing material
        return MexcClient(raw)


    class MexcClient:
        """Concrete MEXC client satisfying the FOUND-08 MexcClient Protocol.

        D-40: ccxt-unified swap futures (BTC/USDT:USDT)
        D-41: ccxt >=4.5.54,<4.6 pin
        D-73: aiolimiter + ccxt internal throttler defense-in-depth
        Pitfall 8: verbose=False (no signing material in logs)
        """

        def __init__(self, raw_client: ccxtpro.mexc) -> None:
            self._client = raw_client

        @mexc_retry
        async def fetch_ohlcv(self, symbol: str, timeframe: str, since: datetime, until: datetime) -> tuple[Candle, ...]:
            since_ms = int(since.timestamp() * 1000)
            async with MEXC_LIMITER:
                try:
                    raw = await self._client.fetch_ohlcv(symbol, timeframe, since=since_ms, limit=1000)
                except Exception as exc:
                    await write_to_dead_letter("mexc_native", "fetch_ohlcv", symbol, "", type(exc).__name__, str(exc))
                    raise
            candles = []
            for row in raw:
                try:
                    s = MexcOhlcvRow.from_ccxt_row(row, symbol=symbol, timeframe=timeframe)
                    candles.append(s.to_domain())
                except Exception as exc:
                    await write_to_dead_letter("mexc_native", "fetch_ohlcv", symbol, str(row), type(exc).__name__, str(exc)[:2000])
            return tuple(candles)

        @mexc_retry
        async def fetch_funding_rate_history(self, symbol: str, since: datetime, until: datetime) -> tuple[Funding, ...]:
            since_ms = int(since.timestamp() * 1000)
            async with MEXC_LIMITER:
                raw = await self._client.fetch_funding_rate_history(symbol, since=since_ms, limit=1000)
            fundings = []
            for d in raw:
                try:
                    f = MexcFundingRow.model_validate({"symbol": symbol, "timestamp": d["timestamp"], "fundingTimestamp": d.get("fundingTimestamp", d["timestamp"]), "fundingRate": str(d["fundingRate"]), "info": d.get("info")})
                    fundings.append(f.to_domain())
                except Exception as exc:
                    await write_to_dead_letter("mexc_native", "fetch_funding_rate_history", symbol, str(d), type(exc).__name__, str(exc)[:2000])
            return tuple(fundings)

        @mexc_retry
        async def fetch_open_interest_history(self, symbol: str, timeframe: str = "1h") -> tuple[MexcOpenInterestRow, ...]:
            async with MEXC_LIMITER:
                raw = await self._client.fetch_open_interest_history(symbol, timeframe)
            rows = []
            for d in raw:
                try:
                    rows.append(MexcOpenInterestRow.model_validate({"symbol": symbol, "timestamp": d["timestamp"], "openInterestAmount": str(d.get("openInterestAmount", "0")), "openInterestValue": str(d.get("openInterestValue", "0")), "info": d.get("info")}))
                except Exception as exc:
                    await write_to_dead_letter("mexc_native", "fetch_open_interest_history", symbol, str(d), type(exc).__name__, str(exc)[:2000])
            return tuple(rows)

        @mexc_retry
        async def fetch_order_book(self, symbol: str, depth: int = 20) -> OrderBook:
            async with MEXC_LIMITER:
                raw = await self._client.fetch_order_book(symbol, limit=depth)
            ts = datetime.fromtimestamp(raw["timestamp"]/1000, tz=timezone.utc) if raw.get("timestamp") else datetime.now(timezone.utc)
            bids = tuple(OrderBookLevel(price=Decimal(str(b[0])), qty=Decimal(str(b[1]))) for b in raw["bids"][:depth])
            asks = tuple(OrderBookLevel(price=Decimal(str(a[0])), qty=Decimal(str(a[1]))) for a in raw["asks"][:depth])
            return OrderBook(symbol=symbol, source="mexc_native", ts=ts, bids=bids, asks=asks)

        async def place_order(self, order: Order) -> str:
            raise NotImplementedError("Phase 5: trade-key required; this is the read-only data-platform client")

        async def cancel_order(self, client_order_id: str) -> None:
            raise NotImplementedError("Phase 5: trade-key required; this is the read-only data-platform client")

        async def close(self) -> None:
            await self._client.close()
    ```

    from collections.abc import Iterable
    import asyncio
    import structlog
    from sqlalchemy.ext.asyncio import AsyncEngine

    from shortfire.ingest.storage.copy import copy_into_hypertable
    from shortfire.ingest.mexc.schemas import MexcOhlcvRow, MexcFundingRow, MexcOpenInterestRow
    from shortfire.ingest.dead_letter.writer import write_to_dead_letter
    from shortfire.clients.mexc import MexcClient as MexcClientProto

    log = structlog.get_logger("ingest.mexc.backfill")

    TIMEFRAME_MS: dict[str, int] = {
        "1m": 60_000,
        "5m": 300_000,
        "15m": 900_000,
        "1h": 3_600_000,
        "4h": 14_400_000,
        "1d": 86_400_000,
    }

    TIMEFRAME_TO_TABLE: dict[str, str] = {
        "1m": "raw_mexc_candles_1m",
        "1d": "raw_mexc_candles_1d",
        # 5m/15m/1h/4h are continuous aggregates per D-67 — NOT backfilled (they materialize from 1m)
    }


    async def backfill_ohlcv(
        engine: AsyncEngine,
        client: MexcClientProto,
        symbol: str,
        timeframe: str,
        since: datetime,
        until: datetime,
        semaphore: asyncio.Semaphore,
    ) -> int:
        """Paginated REST OHLCV backfill — D-42.

        Pagination: each page is `limit=1000`; cursor advances by data[-1][0] + TIMEFRAME_MS[tf]
        Storage: only 1m and 1d timeframes go to typed hypertables (CAs materialize 5m/15m/1h/4h)
        Idempotency: copy_into_hypertable does ON CONFLICT (symbol, ts) DO NOTHING
        """
        if timeframe not in TIMEFRAME_TO_TABLE:
            log.warning("ohlcv.backfill.skipped_ca_timeframe", symbol=symbol, timeframe=timeframe,
                       reason="5m/15m/1h/4h materialize from continuous aggregates over raw_mexc_candles_1m")
            return 0

        target_table = TIMEFRAME_TO_TABLE[timeframe]
        ms_step = TIMEFRAME_MS[timeframe]
        cursor = int(since.timestamp() * 1000)
        end_ms = int(until.timestamp() * 1000)
        total_written = 0

        async with semaphore:
            while cursor < end_ms:
                since_dt = datetime.fromtimestamp(cursor / 1000, tz=timezone.utc)
                until_dt = datetime.fromtimestamp(end_ms / 1000, tz=timezone.utc)
                candles = await client.fetch_ohlcv(symbol, timeframe, since_dt, until_dt)
                if not candles:
                    break
                records = [
                    (c.symbol, c.ts, c.open, c.high, c.low, c.close, c.volume, "mexc_native", "ok")
                    for c in candles
                ]
                n = await copy_into_hypertable(
                    engine,
                    target_table,
                    records,
                    columns=("symbol", "ts", "open", "high", "low", "close", "volume", "source", "quality_flag"),
                    conflict_columns=("symbol", "ts"),
                )
                total_written += n
                last_ts_ms = int(candles[-1].ts.timestamp() * 1000)
                cursor = last_ts_ms + ms_step
        log.info("ohlcv.backfill.completed", symbol=symbol, timeframe=timeframe, total_written=total_written)
        return total_written


    async def backfill_universe(
        engine: AsyncEngine,
        client: MexcClientProto,
        symbols: list[str],
        timeframes: list[str],
        since: datetime,
        until: datetime,
        max_concurrency: int = 8,
    ) -> dict[str, int]:
        """Universe-wide backfill — Semaphore(8) per D-42."""
        sem = asyncio.Semaphore(max_concurrency)
        results: dict[str, int] = {}
        async with asyncio.TaskGroup() as tg:
            tasks: dict[tuple[str, str], asyncio.Task] = {}
            for sym in symbols:
                for tf in timeframes:
                    tasks[(sym, tf)] = tg.create_task(
                        backfill_ohlcv(engine, client, sym, tf, since, until, sem),
                        name=f"backfill.ohlcv.{sym}.{tf}",
                    )
        for (sym, tf), task in tasks.items():
            results[f"{sym}.{tf}"] = task.result()
        return results


    async def backfill_funding(
        engine: AsyncEngine,
        client: MexcClientProto,
        symbol: str,
        since: datetime,
        until: datetime,
        semaphore: asyncio.Semaphore,
    ) -> int:
        """Per-symbol funding history REST backfill — D-44."""
        async with semaphore:
            fundings = await client.fetch_funding_rate_history(symbol, since, until)
            if not fundings:
                return 0
            records = [
                (f.symbol, f.settlement_ts, f.published_ts, f.rate, f.next_funding_ts, "mexc_native", "ok")
                for f in fundings
            ]
            return await copy_into_hypertable(
                engine,
                "raw_mexc_funding",
                records,
                columns=("symbol", "settlement_ts", "published_ts", "funding_rate", "next_funding_ts", "source", "quality_flag"),
                conflict_columns=("symbol", "settlement_ts"),
            )
